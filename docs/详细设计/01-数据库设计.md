# 数据库设计

> 本文档定义 Starcat 的本地数据库 Schema 设计，基于 GRDB/SQLite。

---

## 一、设计原则

### 1.1 数据分类

| 类型 | 说明 | 示例 |
|------|------|------|
| 可重建数据 | 可从 GitHub API 重新拉取 | repos、releases |
| 用户数据 | 用户创建的，不能丢失 | tags、notes、status |
| 缓存数据 | 可按需重新生成 | ai_summaries、embeddings |

### 1.2 性能考虑

- 为常用查询创建索引
- 支持分页和LIMIT
- 使用批量操作提升性能
- FTS5 用于全文搜索

---

## 二、核心表结构

### 2.1 Repo 表（仓库信息）

```sql
CREATE TABLE repos (
    id INTEGER PRIMARY KEY,                -- GitHub repo ID (Int64，唯一标识)
    owner TEXT NOT NULL,                   -- 所有者
    name TEXT NOT NULL,                   -- 仓库名
    full_name TEXT NOT NULL,              -- owner/name 完整名称

    description TEXT,                     -- 描述
    language TEXT,                         -- 主语言
    stars_count INTEGER DEFAULT 0,         -- Star 数
    forks_count INTEGER DEFAULT 0,         -- Fork 数
    watchers_count INTEGER DEFAULT 0,        -- Watcher 数
    topics TEXT,                           -- JSON 数组: ["ai", "swift"]

    license TEXT,                          -- License 名称
    homepage TEXT,                         -- 主页 URL
    html_url TEXT NOT NULL,               -- GitHub 页面 URL
    clone_url TEXT,                       -- HTTPS clone URL
    ssh_url TEXT,                         -- SSH clone URL

    is_private INTEGER DEFAULT 0,         -- 是否私有
    is_fork INTEGER DEFAULT 0,             -- 是否 Fork
    is_archived INTEGER DEFAULT 0,        -- 是否 Archived
    is_starred INTEGER DEFAULT 1,         -- 用户是否 Star 了它

    pushed_at TEXT,                       -- 最后 push 时间 (ISO8601)
    created_at TEXT,                      -- 创建时间
    updated_at TEXT,                      -- 更新时间
    starred_at TEXT,                      -- 用户 star 的时间

    cached_at TEXT                       -- 本地缓存时间
);

CREATE UNIQUE INDEX idx_repos_full_name ON repos(full_name);
CREATE INDEX idx_repos_language ON repos(language);
CREATE INDEX idx_repos_stars ON repos(stars_count);
CREATE INDEX idx_repos_starred_at ON repos(starred_at);
```

### 2.2 StarredRepo 表（用户的 Star 记录）

```sql
CREATE TABLE starred_repos (
    repo_id INTEGER PRIMARY KEY,          -- 关联 repos.id (Int64)
    user_id INTEGER NOT NULL,             -- GitHub 用户 ID (Int64)
    starred_at TEXT NOT NULL,             -- Star 时间

    sync_status TEXT DEFAULT 'synced',   -- synced / pending / failed
    last_sync_at TEXT,                   -- 最后同步时间
    FOREIGN KEY (repo_id) REFERENCES repos(id) ON DELETE CASCADE
);

CREATE INDEX idx_starred_repos_user ON starred_repos(user_id);
CREATE INDEX idx_starred_repos_starred_at ON starred_repos(starred_at);
```

### 2.3 Tag 表（标签）

```sql
CREATE TABLE tags (
    id TEXT PRIMARY KEY,                  -- UUID
    name TEXT NOT NULL,                   -- 标签名
    color TEXT,                           -- 颜色 (hex, 如 #FF5722)
    icon TEXT,                            -- 图标 (SF Symbol 名称)
    sort_order INTEGER DEFAULT 0,        -- 排序顺序
    is_preset INTEGER DEFAULT 0,         -- 是否预设分类 (1=预设)
    parent_id TEXT,                       -- 父标签 ID (嵌套标签)
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,

    FOREIGN KEY (parent_id) REFERENCES tags(id) ON DELETE SET NULL
);

CREATE UNIQUE INDEX idx_tags_name ON tags(name);
CREATE INDEX idx_tags_parent ON tags(parent_id);
```

### 2.4 RepoTag 表（仓库-标签关联）

```sql
CREATE TABLE repo_tags (
    repo_id INTEGER NOT NULL,             -- 关联 repos.id (Int64)
    tag_id TEXT NOT NULL,                  -- 关联 tags.id (UUID)
    created_at TEXT NOT NULL,

    PRIMARY KEY (repo_id, tag_id),
    FOREIGN KEY (repo_id) REFERENCES repos(id) ON DELETE CASCADE,
    FOREIGN KEY (tag_id) REFERENCES tags(id) ON DELETE CASCADE
);

CREATE INDEX idx_repo_tags_tag ON repo_tags(tag_id);
```

### 2.5 RepoNote 表（私有笔记）

```sql
CREATE TABLE repo_notes (
    repo_id INTEGER PRIMARY KEY,          -- 关联 repos.id (Int64)
    content TEXT,                         -- Markdown 格式笔记
    status TEXT DEFAULT 'unread',        -- unread / reading / using / deprecated
    is_ai_generated INTEGER DEFAULT 0,   -- 是否 AI 生成
    edited_at TEXT,                       -- 最后编辑时间
    FOREIGN KEY (repo_id) REFERENCES repos(id) ON DELETE CASCADE
);
```

### 2.6 AISummary 表（AI 摘要）

```sql
CREATE TABLE ai_summaries (
    repo_id INTEGER PRIMARY KEY,          -- 关联 repos.id (Int64)
    one_liner TEXT,                      -- 一句话描述
    summary TEXT,                        -- 中文摘要
    platforms TEXT,                      -- JSON 数组: ["macOS", "iOS"]
    category TEXT,                        -- 预设分类
    pros TEXT,                           -- JSON 数组
    cons TEXT,                           -- JSON 数组
    min_example TEXT,                    -- 最小示例代码
    model_used TEXT,                     -- 使用的模型
    generated_at TEXT,                   -- 生成时间
    FOREIGN KEY (repo_id) REFERENCES repos(id) ON DELETE CASCADE
);
```

### 2.7 SavedSearch 表（保存的搜索）

```sql
CREATE TABLE saved_searches (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    query TEXT NOT NULL,                 -- 搜索条件 JSON
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    last_used_at TEXT
);
```

### 2.8 ReleaseSubscription 表（Release 订阅）

```sql
CREATE TABLE release_subscriptions (
    repo_id INTEGER PRIMARY KEY,          -- 关联 repos.id (Int64)
    is_subscribed INTEGER DEFAULT 1,
    last_check_at TEXT,                  -- 最后检查时间
    last_version TEXT,                   -- 最后知道的版本
    notify_enabled INTEGER DEFAULT 1,
    FOREIGN KEY (repo_id) REFERENCES repos(id) ON DELETE CASCADE
);
```

### 2.9 Release 表（缓存的 Release）

```sql
CREATE TABLE releases (
    id INTEGER PRIMARY KEY,               -- GitHub Release ID (Int64)
    repo_id INTEGER NOT NULL,             -- 关联 repos.id (Int64)
    tag_name TEXT NOT NULL,              -- 版本号
    name TEXT,                           -- Release 名称
    body TEXT,                            -- Release 说明
    is_prerelease INTEGER DEFAULT 0,
    is_draft INTEGER DEFAULT 0,
    published_at TEXT,
    html_url TEXT,

    FOREIGN KEY (repo_id) REFERENCES repos(id) ON DELETE CASCADE
);

CREATE INDEX idx_releases_repo ON releases(repo_id);
CREATE INDEX idx_releases_published ON releases(published_at DESC);
```

### 2.10 SyncState 表（同步状态）

```sql
CREATE TABLE sync_state (
    user_id INTEGER PRIMARY KEY,          -- GitHub 用户 ID (Int64)
    last_sync_at TEXT,                   -- 最后完整同步时间
    last_incremental_at TEXT,             -- 最后增量同步时间
    starred_count INTEGER DEFAULT 0,      -- Star 总数
    synced_count INTEGER DEFAULT 0,       -- 已同步数
    failed_count INTEGER DEFAULT 0,       -- 失败数
    sync_status TEXT DEFAULT 'idle',      -- idle / syncing / failed
    error_message TEXT
);
```

## 三、README 缓存表

```sql
-- README 单独存储，与 repo 表分离
CREATE TABLE readmes (
    repo_id INTEGER PRIMARY KEY,          -- 关联 repos.id (Int64)
    content TEXT,                         -- 原始 Markdown
    rendered_html TEXT,                   -- 渲染后的 HTML（可选）
    etag TEXT,                           -- 用于缓存验证
    last_modified TEXT,                  -- Last-Modified header
    cached_at TEXT NOT NULL,             -- 缓存时间
    size INTEGER DEFAULT 0,              -- README 大小（字节）
    FOREIGN KEY (repo_id) REFERENCES repos(id) ON DELETE CASCADE
);

CREATE INDEX idx_readmes_cached ON readmes(cached_at);
```

## 四、FTS5 全文搜索

> **Tokenizer 选择**：统一使用 `unicode61 remove_diacritics 2`，原因：
> - `unicode61` 对 CJK 字符按字切分（适合中文笔记/描述）
> - `remove_diacritics 2` 处理重音符号（如 `café` → `cafe`），方便英文搜索
> - **不使用 `porter`**：porter 是英文词干算法（`running → run`），对中文反而切碎语义
> - 长期方向：若中文搜索体验不够，可后续替换为 `simple` + jieba 分词或集成 sqlite-vec

```sql
-- 为 Repo 搜索创建的 FTS5 虚拟表（content-less 外部内容模式）
CREATE VIRTUAL TABLE repos_fts USING fts5(
    name,
    description,
    language,
    topics,
    content='repos',
    content_rowid='id',                  -- 直接复用 repos.id（INTEGER）作为 rowid
    tokenize = 'unicode61 remove_diacritics 2'
);

-- 触发器：保持 FTS 与 repos 表同步
CREATE TRIGGER repos_ai AFTER INSERT ON repos BEGIN
    INSERT INTO repos_fts(rowid, name, description, language, topics)
    VALUES (new.id, new.name, new.description, new.language, new.topics);
END;

CREATE TRIGGER repos_ad AFTER DELETE ON repos BEGIN
    INSERT INTO repos_fts(repos_fts, rowid, name, description, language, topics)
    VALUES('delete', old.id, old.name, old.description, old.language, old.topics);
END;

CREATE TRIGGER repos_au AFTER UPDATE ON repos BEGIN
    INSERT INTO repos_fts(repos_fts, rowid, name, description, language, topics)
    VALUES('delete', old.id, old.name, old.description, old.language, old.topics);
    INSERT INTO repos_fts(rowid, name, description, language, topics)
    VALUES (new.id, new.name, new.description, new.language, new.topics);
END;
```

> **说明**：FTS5 外部内容表 `content='repos'` + `content_rowid='id'` 让 FTS 索引与 repos 表零冗余地同步；INSERT/UPDATE/DELETE 触发器维护一致性。Notes 全文搜索如需支持，单独建 `notes_fts` 虚拟表。

## 七、GRDB 模型定义

### 7.1 Repo 模型

```swift
import GRDB

struct Repo: Codable, FetchableRecord, PersistableRecord, Identifiable {
    static let databaseTableName = "repos"

    var id: Int64                          // GitHub repo ID
    var owner: String
    var name: String
    var fullName: String

    var description: String?
    var language: String?
    var starsCount: Int
    var forksCount: Int
    var watchersCount: Int
    var topics: String?  // JSON

    var license: String?
    var homepage: String?
    var htmlUrl: String
    var cloneUrl: String?
    var sshUrl: String?

    var isPrivate: Bool
    var isFork: Bool
    var isArchived: Bool
    var isStarred: Bool

    var pushedAt: String?
    var createdAt: String?
    var updatedAt: String?
    var starredAt: String?

    var cachedAt: String?

    // 关联
    static let starredRepo = hasOne(StarredRepo.self)
    static let note = hasOne(RepoNote.self)
    static let aiSummary = hasOne(AISummary.self)
    static let tags = hasMany(Tag.self, through: hasMany(RepoTag.self), using: RepoTag.tag)
}
```

### 7.2 批量写入

```swift
// 使用 GRDB 的批量写入
try await dbPool.write { db in
    // 批量插入：比单条插入快 10-100 倍
    try Repo.insertAll(repos, onConflict: .replace, using: db)
}

// 或者使用事务
try await dbPool.write { db in
    try db.beginTransaction()
    defer {
        try? db.commit()
    }

    for repo in repos {
        try repo.save(db)
    }
}
```
