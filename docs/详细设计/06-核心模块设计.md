## 一、同步管理器

### 1.1 SyncManager 设计

```swift
/// 同步管理器
/// 使用 @Observable（Swift 5.9+ 可用，不需要 Swift 6 语言模式）
@MainActor
@Observable
class SyncManager {
    var syncState: SyncState = .idle
    var progress: SyncProgress?
    var lastSyncDate: Date?

    private let githubAPI: GitHubAPIClient
    private let database: DatabaseManager
    private let rateLimitHandler: RateLimitHandler

    enum SyncState {
        case idle
        case syncing
        case rateLimited(until: Date)
        case failed(error: Error)
    }

    struct SyncProgress {
        var current: Int
        var total: Int
        var currentRepo: String?
    }
}
```

### 1.2 同步流程

```swift
func performSync() async {
    guard syncState != .syncing else { return }

    syncState = .syncing
    progress = SyncProgress(current: 0, total: 0, currentRepo: nil)

    do {
        // 1. 检查网络
        guard await NetworkMonitor.shared.isConnected else {
            syncState = .failed(error: NetworkError.noConnection)
            return
        }

        // 2. 获取用户 stars 总数
        let totalCount = try await githubAPI.getStarredCount()
        progress?.total = totalCount

        // 3. 分页同步
        var page = 1
        var synced = 0
        var hasMore = true

        while hasMore {
            // 检查 Rate Limit
            if let retryAfter = await checkRateLimit() {
                syncState = .rateLimited(until: Date().addingTimeInterval(retryAfter))
                try await Task.sleep(nanoseconds: UInt64(retryAfter * 1_000_000_000))
                syncState = .syncing
            }

            progress?.current = synced
            progress?.currentRepo = "Page \(page)"

            // 4. 拉取一页数据
            let repos = try await githubAPI.starredRepos(page: page, perPage: 100)

            // 5. 写入数据库
            try await database.upsertRepos(repos)

            synced += repos.count
            progress?.current = synced

            // 6. 检查是否完成
            hasMore = repos.count == 100
            page += 1

            // 7. 保存进度
            try await saveSyncProgress(page: page, synced: synced)
        }

        // 8. 完成
        lastSyncDate = Date()
        syncState = .idle
        progress = nil

    } catch {
        syncState = .failed(error: error)
    }
}
```

### 1.3 后台同步

```swift
// macOS 使用 NSBackgroundActivityScheduler 进行后台同步
// 注意：BGAppRefreshTask / BGTaskScheduler 是 iOS/tvOS 专用 API
func scheduleBackgroundSync() {
    let scheduler = NSBackgroundActivityScheduler(identifier: "com.starchat.app.sync")
    scheduler.repeats = true
    scheduler.interval = 60 * 60 // 1小时

    scheduler.schedule { completion in
        Task {
            await self.performSync()
            completion(.finished)
        }
    }
}
```

> **平台差异**：
> - iOS/tvOS：使用 `BGTaskScheduler` + `BGAppRefreshTask`
> - macOS：使用 `NSBackgroundActivityScheduler`

### 2.3 缓存策略

```swift
actor AIServiceClient: AIServiceProtocol {
    private let remote: AIProxyClient
    private let cache: AICache

    func summarize(repo: RepoInfo, readme: String, options: SummaryOptions?) async throws -> AISummary {
        // 1. 检查缓存
        if let cached = await cache.getSummary(repoId: repo.id) {
            return cached
        }

        // 2. 调用远程
        let summary = try await remote.summarize(repo: repo, readme: readme, options: options)

        // 3. 存入缓存
        await cache.saveSummary(repo.id, summary)

        return summary
    }
}
```

## 三、Keychain 管理

### 3.1 KeychainManager

```swift
class KeychainManager: Sendable {
    static let shared = KeychainManager()

    private let service = "com.starchat.app"

    // GitHub Token
    func storeGithubToken(_ token: String) throws {
        try store(key: "github_access_token", data: Data(token.utf8))
    }

    func getGithubToken() -> String? {
        guard let data = getData(key: "github_access_token") else { return nil }
        return String(data: data, encoding: .utf8)
    }

    func deleteGithubToken() throws {
        try delete(key: "github_access_token")
    }

    // AI API Key
    func storeAIKey(_ key: String) throws {
        try store(key: "ai_api_key", data: Data(key.utf8))
    }

    func getAIKey() -> String? {
        guard let data = getData(key: "ai_api_key") else { return nil }
        return String(data: data, encoding: .utf8)
    }
}
```

### 3.2 安全存储实现

```swift
private func store(key: String, data: Data) throws {
    let query: [String: Any] = [
        kSecClass as String: kSecClassGenericPassword,
        kSecAttrService as String: service,
        kSecAttrAccount as String: key,
        kSecValueData as String: data,
        kSecAttrAccessible as String: kSecAttrAccessibleAfterFirstUnlock
    ]

    // 先删除已存在的
    SecItemDelete(query as CFDictionary)

    // 再添加新的
    let status = SecItemAdd(query as CFDictionary, nil)
    guard status == errSecSuccess else {
        throw KeychainError.unableToStore
    }
}

private func getData(key: String) -> Data? {
    let query: [String: Any] = [
        kSecClass as String: kSecClassGenericPassword,
        kSecAttrService as String: service,
        kSecAttrAccount as String: key,
        kSecReturnData as String: true
    ]

    var result: AnyObject?
    let status = SecItemCopyMatching(query as CFDictionary, &result)

    guard status == errSecSuccess else { return nil }
    return result as? Data
}
```

```swift
enum LogLevel: String {
    case debug = "DEBUG"
    case info = "INFO"
    case warning = "WARNING"
    case error = "ERROR"
}

struct Logger {
    static let shared = Logger()

    private let subsystem = "com.starchat.app"
    private let osLog: OSLog

    init(category: String = "general") {
        osLog = OSLog(subsystem: subsystem, category: category)
    }

    func log(level: LogLevel, _ message: String) {
        os_log("%{public}@", log: osLog, type: osLogType(level), message)
    }

    private func osLogType(_ level: LogLevel) -> OSLogType {
        switch level {
        case .debug: return .debug
        case .info: return .info
        case .warning: return .default
        case .error: return .error
        }
    }
}
```

### 5.2 错误恢复

```swift
@MainActor
@Observable
class ErrorHandler {
    var currentError: AppError?
    var showError = false

    func handle(_ error: Error) {
        currentError = mapError(error)
        showError = true
    }

    private func mapError(_ error: Error) -> AppError {
        switch error {
        case let error as NetworkError:
            return .network(error)
        case let error as DatabaseError:
            return .database(error)
        case let error as AuthError:
            return .auth(error)
        case let error as AIServiceError:
            return .ai(error)
        default:
            return .unknown(error)
        }
    }

    func recoveryAction(for error: AppError) -> RecoveryAction? {
        switch error {
        case .network(.noConnection):
            return RecoveryAction(
                title: "检查网络",
                action: { /* 打开设置 */ }
            )
        case .auth(.tokenExpired):
            return RecoveryAction(
                title: "重新登录",
                action: { /* 引导重新登录 */ }
            )
        case .network(.rateLimited):
            return RecoveryAction(
                title: "稍后重试",
                action: { /* 等待后重试 */ }
            )
        default:
            return nil
        }
    }
}

struct RecoveryAction {
    let title: String
    let action: () -> Void
}
```

## 六、Release 通知监测

### 6.1 ReleaseMonitor 设计

```swift
/// Release 发布通知监测管理器
/// 定期检查用户订阅的仓库是否有新 Release 发布
@MainActor
@Observable
class ReleaseMonitor {
    var newReleases: [ReleaseNotification] = []
    var isChecking = false

    private let githubAPI: GitHubAPIClient
    private let database: DatabaseManager

    /// 检查所有订阅的仓库是否有新 Release
    func checkForNewReleases() async {
        // 1. 获取所有订阅的仓库
        let subscriptions = database.getActiveSubscriptions()

        for subscription in subscriptions {
            // 2. 检查是否有新 Release
            if let newRelease = await checkRepo(subscription) {
                newReleases.append(newRelease)
            }
        }
    }

    /// 检查单个仓库的最新 Release
    private func checkRepo(_ subscription: ReleaseSubscription) async -> ReleaseNotification? {
        do {
            let latestRelease = try await githubAPI.latestRelease(
                owner: subscription.repoOwner,
                repo: subscription.repoName
            )

            // 比较版本
            if latestRelease.tagName != subscription.lastKnownVersion {
                return ReleaseNotification(
                    repo: subscription.repo,
                    release: latestRelease,
                    isNew: true
                )
            }
        } catch {
            // 忽略错误，下次再试
        }
        return nil
    }
}

/// Release 通知
struct ReleaseNotification: Identifiable {
    let id = UUID()
    let repo: Repo
    let release: Release
    let receivedAt: Date = Date()
}
```

### 6.3 通知推送

```swift
/// 使用 UNUserNotificationCenter 推送通知
func sendReleaseNotification(_ release: ReleaseNotification) async {
    let content = UNMutableNotificationContent()
    content.title = "\(release.repo.name) 发布新版本"
    content.body = "\(release.release.tagName): \(release.release.name ?? "查看详情")"
    content.sound = .default
    content.userInfo = [
        "repoId": release.repo.id,
        "releaseUrl": release.release.htmlUrl
    ]

    let request = UNNotificationRequest(
        identifier: "release-\(release.repo.id)-\(release.release.id)",
        content: content,
        trigger: nil
    )

    try? await UNUserNotificationCenter.current().add(request)
}
```
